Per RESEARCH.md §Headline Derivation Algorithm + UI-SPEC #copywriting, the algorithm is:

1. **Tier 1** — Active USB-C charging port: filter `category === 'TypeCPort' && status === 'Charging'`. For each, parse wattage + direction from `bullets[]`. Rank by wattage desc, tie-break by lowest `port_number`. Format per direction:
   - `sink`   -> `Charging: {watts} in`
   - `source` -> `Powering: {watts} out`
   - unknown direction, non-zero watts -> `USB-C: {watts}`
   - unknown direction, zero watts -> `USB-C: charging`
2. **Tier 2** — Fastest attached non-charging link: any device with parseable link speed > 0 bps. Sort by bps desc, tie-break by `id.localeCompare`. Format: `{usbVersionLabel} · {humanRate}` (middle dot is U+00B7, NOT a hyphen-minus).
3. **Tier 3** — Devices attached with no link-speed data: count entries with `status !== 'Empty'`. Format: `1 device` / `{N} devices`.
4. **Tier 4** — Empty: `Nothing connected`.

Watts formatting (`formatWatts`): integer if >= 10 W, one decimal if < 10 W (RESEARCH §Headline Derivation).

Full file body (paste-ready):

```javascript
import GObject from 'gi://GObject';

function unpackDeviceEntry(tuple) {
    return {
        id:          tuple[0],
        category:    tuple[1],
        status:      tuple[2],
        headline:    tuple[3],
        subtitle:    tuple[4],
        icon:        tuple[5],
        bullets:     tuple[6],
        port_number: tuple[7],
    };
}

const WATT_RE        = /(\d+(?:\.\d+)?)\s*W\b/i;
const DIRECTION_RE   = /\b(sink|source)\b/i;
const USB_VERSION_RE = /\b(USB\s+\d+(?:\.\d+)?(?:\s+Gen\s+\d+(?:x\d+)?)?)/i;
const SPEED_RE       = /(\d+(?:\.\d+)?)\s*(Gb|Mb|Kb)\/s/i;

function parseWatts(bullets) {
    for (const b of bullets || []) {
        const m = WATT_RE.exec(b);
        if (m) return parseFloat(m[1]);
    }
    return 0;
}

function parseDirection(bullets) {
    for (const b of bullets || []) {
        const m = DIRECTION_RE.exec(b);
        if (m) return m[1].toLowerCase();
    }
    return null;
}

function parseLinkSpeed(bullets) {
    let usbVersionLabel = '';
    let humanRate = '';
    let bps = 0;
    for (const b of bullets || []) {
        if (!usbVersionLabel) {
            const v = USB_VERSION_RE.exec(b);
            if (v) usbVersionLabel = v[1].replace(/\s+/g, ' ');
        }
        if (!humanRate) {
            const s = SPEED_RE.exec(b);
            if (s) {
                const n = parseFloat(s[1]);
                const unit = s[2].toLowerCase();
                humanRate = `${s[1]} ${s[2]}/s`;
                bps = unit === 'gb' ? n * 1e9 : unit === 'mb' ? n * 1e6 : n * 1e3;
            }
        }
        if (usbVersionLabel && humanRate) break;
    }
    return {bps, usbVersionLabel, humanRate};
}

function formatWatts(w) {
    return w >= 10 ? `${Math.round(w)} W` : `${w.toFixed(1)} W`;
}

export function deriveSubtitle(devices) {
    const ports = devices.filter(d =>
        d.category === 'TypeCPort' && d.status === 'Charging');
    if (ports.length > 0) {
        const ranked = ports
            .map(p => ({
                port:      p,
                watts:     parseWatts(p.bullets),
                direction: parseDirection(p.bullets),
            }))
            .sort((a, b) => b.watts - a.watts
                         || a.port.port_number - b.port.port_number);
        const top = ranked[0];
        if (top.direction === 'sink')
            return `Charging: ${formatWatts(top.watts)} in`;
        if (top.direction === 'source')
            return `Powering: ${formatWatts(top.watts)} out`;
        return top.watts > 0
            ? `USB-C: ${formatWatts(top.watts)}`
            : 'USB-C: charging';
    }

    const withSpeed = devices
        .map(d => ({device: d, speed: parseLinkSpeed(d.bullets)}))
        .filter(x => x.speed.bps > 0 && x.speed.usbVersionLabel && x.speed.humanRate);
    if (withSpeed.length > 0) {
        withSpeed.sort((a, b) => b.speed.bps - a.speed.bps
                              || a.device.id.localeCompare(b.device.id));
        const top = withSpeed[0];
        return `${top.speed.usbVersionLabel} · ${top.speed.humanRate}`;
    }

    const attached = devices.filter(d => d.status !== 'Empty');
    if (attached.length > 0) {
        return attached.length === 1 ? '1 device' : `${attached.length} devices`;
    }

    return 'Nothing connected';
}

export const DeviceStore = GObject.registerClass({
    Signals: {'changed': {}},
}, class DeviceStore extends GObject.Object {
    constructor() {
        super();
        this._devices = [];
        this._daemonRunning = false;
    }

    get devices()        { return this._devices; }
    get daemonRunning()  { return this._daemonRunning; }

    get subhead() {
        if (!this._daemonRunning) return 'Daemon not running';
        return deriveSubtitle(this._devices);
    }

    setDevices(rawEntries) {
        this._devices = (rawEntries || []).map(unpackDeviceEntry);
        this.emit('changed');
    }

    setDaemonRunning(running) {
        if (this._daemonRunning === running) return;
        this._daemonRunning = running;
        this.emit('changed');
    }
});
```

```javascript
_scheduleRefresh() {
    // D-10: 150 ms trailing-edge debounce.
    // Single shared GLib.timeout_add source — reset on every incoming signal
    // so a burst of N signals within 150 ms collapses to exactly 1 snapshot.
    if (this._debounceId !== 0) {
        GLib.Source.remove(this._debounceId);
        this._debounceId = 0;
        // Note: the old SignalRegistry entry's dispose-fn will call
        // GLib.Source.remove on the now-removed id; GLib treats this as a
        // no-op (it just returns false). PITFALLS.md §9 documents this as safe.
    }
    this._debounceId = GLib.timeout_add(
        GLib.PRIORITY_DEFAULT, 150, () => {
            this._debounceId = 0;
            this._snapshotImmediate();
            return GLib.SOURCE_REMOVE;
        });
    // Track the new timer so disable() removes it if it's still in flight
    // (RESEARCH §Validation 23 — disable while debounce pending).
    this._registry.addTimeout(this._debounceId);
}
```

```javascript
export function populateDeviceRows(section, store) {
    section.removeAll();
    if (store.devices.length === 0) {
        section.addMenuItem(new PopupMenu.PopupMenuItem(
            'No USB devices attached',
            {reactive: false, can_focus: false},
        ));

    }
}

export function populateEmptyState(section) {
    section.removeAll();

function buildDeviceRow(device) {
    // Headline as the PopupMenuItem.label (top line, picks up Shell styling
    // for .popup-menu-item .label — UI-SPEC #typography 600-weight row headline).
    // Use device.headline verbatim; fall back to device.id only if headline is
    // empty (shouldn't happen — daemon always emits a headline per RESEARCH
    // §Daemon Wire Shape).
    const headline = device.headline || device.id || '';
    const item = new PopupMenu.PopupMenuItem(headline, {
        reactive: false,
        can_focus: false,
        style_class: 'usbee-device-row',
    });

    // Body box — one St.Label per bullet, line-wrapped. Picks up the
    // .usbee-device-row .body StLabel + StLabel { padding-top: 4px; } rule
    // from stylesheet.css (UI-SPEC #spacing 4px token).
    const body = new St.BoxLayout({
        vertical: true,
        x_expand: true,
        style_class: 'body',
    });

        lbl.clutter_text.line_wrap = true;
        lbl.clutter_text.line_wrap_mode = 2; // Pango.WrapMode.WORD_CHAR
        body.add_child(lbl);
    }
    item.add_child(body);
    return item;
}
```
